cmake_minimum_required(VERSION 3.21)
project(tree_md VERSION 1.0.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_AUTOMOC ON)
set(CMAKE_AUTOUIC ON)
set(CMAKE_AUTORCC ON)
find_package(Qt6 6.2 REQUIRED COMPONENTS Widgets Network)
add_library(tree_md_core STATIC core/MarkdownFileIO.cpp core/MarkdownNode.cpp core/MarkdownManager.cpp core/MarkdownParser.cpp core/MarkdownRelation.cpp core/FileWatcher.cpp)
target_sources(tree_md_core PRIVATE core/MarkdownImages.cpp core/ImageLoader.cpp core/ImageStorage.cpp core/MarkdownExportJob.cpp)
target_sources(tree_md_core PRIVATE core/KnowledgeIndex.cpp)
target_sources(tree_md_core PRIVATE core/DefaultLibrary.cpp)
target_link_libraries(tree_md_core PUBLIC Qt6::Core Qt6::Gui Qt6::Network)
target_include_directories(tree_md_core PUBLIC ${CMAKE_CURRENT_SOURCE_DIR})
add_library(tree_md_ui STATIC ui/MainWindow.cpp ui/MarkdownEditor.cpp ui/MarkdownDocumentView.cpp ui/ImageInput.cpp ui/KnowledgeGraphWidget.cpp ui/KnowledgeNodeWidget.cpp ui/DocumentSidebar.cpp ui/OutlineSidebar.cpp ui/RelationDialog.cpp ui/Theme.cpp)
target_sources(tree_md_ui PRIVATE ui/MarkdownSourceMap.cpp ui/AppIcon.cpp resources/branding.qrc)
target_sources(tree_md_ui PRIVATE ui/MarkdownTypography.cpp resources/typography.qrc)
target_sources(tree_md_ui PRIVATE ui/KnowledgeTreeCanvas.cpp ui/KnowledgeTreeLayout.cpp)
target_link_libraries(tree_md_ui PUBLIC tree_md_core Qt6::Widgets)
qt_add_executable(tree_md main.cpp)
if(WIN32)
    enable_language(RC)
    configure_file(resources/tree_md.rc ${CMAKE_CURRENT_BINARY_DIR}/tree_md-version.rc @ONLY)
    target_sources(tree_md PRIVATE ${CMAKE_CURRENT_BINARY_DIR}/tree_md-version.rc)
    target_include_directories(tree_md PRIVATE ${CMAKE_CURRENT_SOURCE_DIR}/resources)
endif()
target_link_libraries(tree_md PRIVATE tree_md_ui)
set_target_properties(tree_md PROPERTIES WIN32_EXECUTABLE TRUE)
target_compile_definitions(tree_md PRIVATE TREE_MD_SOURCE_DIR="${CMAKE_CURRENT_SOURCE_DIR}" TREE_MD_VERSION="${PROJECT_VERSION}")
install(TARGETS tree_md RUNTIME DESTINATION .)
install(DIRECTORY examples/md_data DESTINATION .)
install(DIRECTORY examples/md_photo DESTINATION .)
include(CTest)
if(BUILD_TESTING)
    find_package(Qt6 REQUIRED COMPONENTS Test)
    add_executable(tree_md_tests tests/TreeMdTests.cpp)
    target_link_libraries(tree_md_tests PRIVATE tree_md_ui Qt6::Test)
    add_test(NAME tree_md_regressions COMMAND tree_md_tests)
    add_executable(tmd_library_tests tests/DefaultLibraryTests.cpp)
    target_link_libraries(tmd_library_tests PRIVATE tree_md_core Qt6::Test)
    add_test(NAME tmd_library_initialization COMMAND tmd_library_tests)
    set_tests_properties(tree_md_regressions PROPERTIES TIMEOUT 120 ENVIRONMENT "QT_QPA_PLATFORM=offscreen;QT_QPA_FONTDIR=C:/Windows/Fonts")
endif()
